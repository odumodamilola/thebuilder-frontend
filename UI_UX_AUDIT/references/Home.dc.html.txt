<script type="text/x-dc" data-dc-script data-props="{&quot;loading&quot;:{&quot;editor&quot;:&quot;boolean&quot;,&quot;default&quot;:false,&quot;tsType&quot;:&quot;boolean&quot;}}">
class Component extends DCLogic {
  state = { w: window.innerWidth };
  componentDidMount() {
    this.r = () => this.setState({ w: window.innerWidth }); window.addEventListener('resize', this.r);
    this.seen = !!window.TBH; if (window.TBH) { if (!this.rendered) this.forceUpdate(); } else this.t = setInterval(() => { if (window.TBH) { clearInterval(this.t); this.forceUpdate(); } }, 50);
  }
  componentWillUnmount() { window.removeEventListener('resize', this.r); clearInterval(this.t); cancelAnimationFrame(this.raf); }
  scrollerRef = (el) => { this.el = el; };
  reduce() { return window.TBH ? window.TBH.reduce() : false; }
  scrollBy(d) {
    const el = this.el; if (!el) return;
    const first = el.querySelector('[style*="scroll-snap-align"]');
    const step = (first ? first.getBoundingClientRect().width : 296) + 20;
    el.scrollBy({ left: d * step, behavior: this.reduce() ? 'auto' : 'smooth' });
  }
  // Mouse drag with momentum; touch uses native swipe momentum
  dragStart = (e) => {
    if (e.pointerType !== 'mouse' || e.button !== 0 || !this.el) return;
    cancelAnimationFrame(this.raf);
    this.drag = { x: e.clientX, left: this.el.scrollLeft, lx: e.clientX, lt: performance.now(), v: 0, moved: false, id: e.pointerId };
  };
  dragMove = (e) => {
    const d = this.drag; if (!d) return;
    const dx = e.clientX - d.x;
    if (!d.moved && Math.abs(dx) > 5) { d.moved = true; this.el.style.scrollSnapType = 'none'; this.el.style.cursor = 'grabbing'; try { this.el.setPointerCapture(d.id); } catch (er) {} }
    if (!d.moved) return;
    this.el.scrollLeft = d.left - dx;
    const now = performance.now(), dt = Math.max(1, now - d.lt);
    d.v = 0.8 * ((d.lx - e.clientX) / dt) + 0.2 * d.v; d.lx = e.clientX; d.lt = now;
  };
  dragEnd = () => {
    const d = this.drag; if (!d) return; this.drag = null;
    const el = this.el; el.style.cursor = 'grab'; this.justDragged = d.moved;
    const done = () => { el.style.scrollSnapType = 'x mandatory'; };
    let v = d.v * 16;
    if (!d.moved || this.reduce() || Math.abs(v) < 0.5) return done();
    const step = () => { el.scrollLeft += v; v *= 0.94; if (Math.abs(v) > 0.4) this.raf = requestAnimationFrame(step); else done(); };
    this.raf = requestAnimationFrame(step);
  };
  dragClick = (e) => { if (this.justDragged) { e.preventDefault(); e.stopPropagation(); this.justDragged = false; } };
  renderVals() {
    const H = window.TBH; this.rendered = !!H; if (!H) return {};
    const mobile = H.bp(this.state.w) === 'm';
    const loading = !!this.props.loading;
    return {
      loading, ready: !loading, notMobile: !mobile, noop: H.noop, sk4: [1, 2, 3, 4],
      col: mobile ? '78%' : 'minmax(260px, 296px)',
      startups: H.STARTUPS, cardProps: { style: { width: '100%' } },
      scrollerRef: this.scrollerRef,
      prev: () => this.scrollBy(-1), next: () => this.scrollBy(1),
      dragStart: this.dragStart, dragMove: this.dragMove, dragEnd: this.dragEnd, dragClick: this.dragClick,
    };
  }
}
